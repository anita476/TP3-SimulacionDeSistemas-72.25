\section{Introducción}

\begin{frame}{Modelo}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \textbf{Evolución temporal}\par\vspace{4pt}
      \setlength{\abovedisplayskip}{0pt}\setlength{\abovedisplayshortskip}{0pt}
      \begin{align}
        \vv{x}_i(t+1) &= \vv{x}_i(t) + \vv{v}_i(t)\,\Delta t \\[2pt]
        \theta(t+1) &= \langle \theta(t) \rangle_{r_{c}} + \Delta\theta \\[2pt]
        \theta(t+1) &= \theta_j(t) + \Delta\theta
      \end{align}
    \end{column}
    \begin{column}{0.42\textwidth}
    
      \textbf{}\par\vspace{4pt}
      {\footnotesize
      Ec.~(2), promedio de Vicsek:
      \[ \langle \theta(t) \rangle_{r_{c}} = \operatorname{atan2}\!\left[
         \langle \sin\theta(t) \rangle_{r_{c}},\ \langle \cos\theta(t) \rangle_{r_{c}} \right] \]
      sobre las partículas dentro de $r_{c}$, incluyendo la propia.

      \vspace{4pt}
      Ec.~(3), votante: $j$ vecino elegido al azar.

      \vspace{4pt}
      $\Delta\theta$: ruido uniforme en $[-\eta/2,\ \eta/2]$.\par}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Paso de simulación}
  \begin{center}
    \begin{minipage}{0.85\textwidth}
      \scriptsize
      \begin{algorithmic}[1]

        \For{$t = 0,\ldots,T-1$}

          \State $\mathcal{V} \gets$ calcular vecinos mediante CIM

          \For{$i = 1,\ldots,N$}

            \If{el modelo es Vicsek}
              \State $\bar{\theta} \gets
              \operatorname{atan2}\left(
              \sum\limits_{j \in \mathcal{V}_i \cup \{i\}}
              \sin(\theta_j),
              \sum\limits_{j \in \mathcal{V}_i \cup \{i\}}
              \cos(\theta_j)
              \right)$
            \EndIf

            \If{el modelo es votante}
              \State elegir $k \in \mathcal{V}_i$ al azar
              \State $\bar{\theta} \gets \theta_k$
            \EndIf

            \State $\theta_i^{\mathrm{nuevo}}
            \gets \bar{\theta} + \Delta\theta$

          \EndFor

          \For{$i = 1,\ldots,N$}
            \State $\vv{x}_i \gets
            \left(\vv{x}_i + \vv{v}_i \Delta t\right) \bmod L$
            \State $\theta_i \gets \theta_i^{\mathrm{nuevo}}$
          \EndFor

        \EndFor

      \end{algorithmic}
    \end{minipage}
  \end{center}
\end{frame}

      \end{parametros}

\begin{frame}{Observables}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \textbf{Observables primarios}\par\vspace{6pt}
      \small
      \begin{itemize}\setlength{\itemsep}{12pt}
        \item Parámetro de orden:
              \[ \va(t) = \frac{1}{N\,v}\left| \sum_{i=1}^{N} \vv{v}_i(t) \right| \]
              {\footnotesize 0: desorden total. \ 1: polarización total.}
        \item Fracción en la componente gigante:
              \[ S(t) = \frac{\left| \mathcal{C}(t) \right|}{N} \]
              {\footnotesize $\mathcal{C}$: mayor cluster; vecinos a distancia $< r_{c}$
              encadenados.}
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \textbf{Observables escalares}\par\vspace{6pt}
      \small
      \begin{itemize}\setlength{\itemsep}{12pt}
        \item Promedio en el estacionario, dentro de cada realización:
              \[ \langle \va \rangle_{\text{realiz.}} =
                 \big\langle\, \va(t) \,\big\rangle_{t \geq t^{*}} \]
        \item Promedio entre las realizaciones:
              \[ \langle \va \rangle = \big\langle\, \langle \va \rangle_{\text{realiz.}}
                 \,\big\rangle_{10\ \text{realiz.}} \]
        \item Ídem con $S(t)$
      \end{itemize}
      \vspace{6pt}
      {\footnotesize Barra de error: desvío entre totalidad de puntos a partir t^{*}
      \par
      $t^{*}$ se fija en relación a la evolución temporal}
    \end{column}
  \end{columns}
\end{frame}


\begin{frame}{Observables: cálculo de $S$}
    \centering{
      \scriptsize
      \begin{algorithmic}[1]
        \State Obtener las posiciones de las partículas en el tiempo $t$
        \State Construir la lista de vecinos $vecinos[i]$

        \State $clusters \gets \emptyset$
        \State Marcar todas las partículas como no visitadas

        \ForAll{partícula $i$ no visitada}
          \State Crear un cluster vacío
          \State Explorar $i$ y sus vecinos
          \State Agregar al cluster todas las partículas conectadas
          \State Marcar las partículas como visitadas
          \State Agregar el cluster a $clusters$
        \EndFor

        \State $N \gets$ cantidad total de partículas
        \State $N_{\max} \gets$ tamaño del cluster más grande
        \State $S(t) \gets N_{\max}/N$

        \State \Return $clusters$, $S(t)$
      \end{algorithmic}
      }
\end{frame}

\begin{frame}{Conclusiones}
  \begin{itemize}\setlength{\itemsep}{14pt}
\item \textbf{Mayor ruido decrece} observable de forma monótona y continua.

\item \textbf{$v_a$}: mayor densidad \textbf{aumenta} observable en Vicsek, \textbf{disminuye} en votante. 

\item \textbf{$S$}: comportamiento similar, pero convergencia a mismo valor (distintas $\rho$). 

\item $v_a$ y $S$: mismo ruido, mayor $v_a$ para Vicsek.

\item \textbf{Vicsek: mayor tolerancia al ruido} para $v_a$ y $S$ (menor diferencia).

\item Ejecuciones CIM: 78\% parámetros por defecto, \textbf{20\% para mismas condiciones} (peor caso).

  \end{itemize}
